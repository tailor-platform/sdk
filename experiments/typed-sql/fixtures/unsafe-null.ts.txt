import { sql } from "@typed-sql/postgres";
import type { QueryRow } from "@typed-sql/core";

export const query = sql`
  SELECT a."email", p."title"
  FROM "Account" AS a
  LEFT JOIN "Project" AS p ON p."ownerId" = a."id"
`;

export function consume(row: QueryRow<typeof query>): string {
  return row.title.toUpperCase();
}
